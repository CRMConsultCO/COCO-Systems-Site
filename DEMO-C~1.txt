function submitBooking(){
  var values = {
    firstName: document.getElementById('firstName').value.trim(),
    lastName: document.getElementById('lastName').value.trim(),
    phone: document.getElementById('phone').value.trim(),
    email: document.getElementById('email').value.trim(),
    vehYear: document.getElementById('vehYear').value.trim(),
    vehMake: document.getElementById('vehMake').value.trim(),
    vehModel: document.getElementById('vehModel').value.trim(),
    vin: document.getElementById('vin').value.trim(),
    service: document.getElementById('service').value.trim(),
    details: document.getElementById('details').value.trim(),
    contactMethod: document.getElementById('contactMethod').value,
    budget: document.getElementById('budget').value,
    apptDate: document.getElementById('apptDate').value,
    apptTime: document.getElementById('apptTime').value
  };

  var required = ['firstName','phone','vehYear','vehMake','vehModel','service'];
  var ok = required.every(function(k){ return values[k]; });
  if(!ok){ document.getElementById('err').style.display = 'block'; return; }
  document.getElementById('err').style.display = 'none';
  document.getElementById('confirm').style.display = 'block';

  var curtain = document.getElementById('curtain');
  curtain.style.display = 'block';
  curtain.textContent = '> RECEIVING SUBMISSION...';
  setTimeout(function(){ curtain.textContent += '\n> WRITING RECORD TO DATABASE...'; }, 400);
  setTimeout(function(){ curtain.textContent += '\n> CREATING CLIENT + VEHICLE RECORD...'; }, 800);
  setTimeout(function(){ curtain.textContent += '\n> ROUTING TO TRELLO — CHECKED IN / INTAKE...'; }, 1200);
  setTimeout(function(){
    curtain.textContent += '\n> DONE. Opening the production board...';
    try{
      window.parent.postMessage({ type: 'chopshop-ticket', data: values }, '*');
    }catch(err){}
  }, 1700);
}
function toggleCurtain(){
  var c = document.getElementById('curtain');
  if(c.textContent.trim() === '') return;
  c.style.display = c.style.display === 'block' ? 'none' : 'block';
}
