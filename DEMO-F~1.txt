<script>
function submitBooking(){
  var ids = ['name','contact','need','type'];
  var values = {};
  var ok = true;
  ids.forEach(function(id){
    var v = document.getElementById(id).value.trim();
    if(!v) ok = false;
    values[id] = v;
  });
  if(!ok){ document.getElementById('err').style.display = 'block'; return; }
  document.getElementById('err').style.display = 'none';
  document.getElementById('confirm').style.display = 'block';

  var curtain = document.getElementById('curtain');
  curtain.style.display = 'block';
  curtain.textContent = '> RECEIVING SUBMISSION...';
  setTimeout(function(){ curtain.textContent += '\n> WRITING RECORD TO DATABASE...'; }, 500);
  setTimeout(function(){ curtain.textContent += '\n> ROUTING TO TRELLO, CALENDAR, NOTION...'; }, 1000);
  setTimeout(function(){
    curtain.textContent += '\n> DONE. Logged across every connected piece:\n\n' +
      'Name: ' + values.name + '\nContact: ' + values.contact +
      '\nNeeds: ' + values.need + '\nType: ' + values.type;
  }, 1600);
}
function toggleCurtain(){
  var c = document.getElementById('curtain');
  if(c.textContent.trim() === '') return;
  c.style.display = c.style.display === 'block' ? 'none' : 'block';
}
</script>
